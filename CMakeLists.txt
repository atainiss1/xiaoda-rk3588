cmake_minimum_required(VERSION 3.16)
project(rk3588_voice VERSION 0.2.0 LANGUAGES C CXX)
find_package(Threads REQUIRED)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_executable(rkvoice_pcm src/pcm_main.cpp)
target_compile_options(rkvoice_pcm PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
option(RKVOICE_BUILD_AUDIO "Build PulseAudio capture/playback CLI" ON)
if(RKVOICE_BUILD_AUDIO)
  add_executable(rkvoice_audio src/audio_main.cpp)
  target_compile_options(rkvoice_audio PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
  if(CMAKE_CROSSCOMPILING)
    set(RKVOICE_PULSE_SDK "${CMAKE_SOURCE_DIR}/build/pulse-sdk" CACHE PATH "Board-matched ARM64 PulseAudio SDK")
    set(PULSE_LIB_DIR "${RKVOICE_PULSE_SDK}/usr/lib/aarch64-linux-gnu")
    if(NOT EXISTS "${RKVOICE_PULSE_SDK}/usr/include/pulse/simple.h")
      message(FATAL_ERROR "Run bash scripts/prepare-pulse-sdk.sh first")
    endif()
    target_include_directories(rkvoice_audio SYSTEM PRIVATE "${RKVOICE_PULSE_SDK}/usr/include")
    target_link_libraries(rkvoice_audio PRIVATE "${PULSE_LIB_DIR}/libpulse-simple.so" "${PULSE_LIB_DIR}/libpulse.so")
    # Only this library subset is staged here. Transitive runtime dependencies
    # are supplied by the board's installed, matching Ubuntu packages; verify
    # with ldd -r on the board. Direct executable symbols are still resolved.
    target_link_options(rkvoice_audio PRIVATE "-Wl,--allow-shlib-undefined")
    set_target_properties(rkvoice_audio PROPERTIES SKIP_BUILD_RPATH TRUE)
  else()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(PULSE REQUIRED IMPORTED_TARGET libpulse-simple)
    target_link_libraries(rkvoice_audio PRIVATE PkgConfig::PULSE)
  endif()
endif()
option(RKVOICE_BUILD_CLOUD "Build Xiaozhi C++ cloud voice client" ON)
if(RKVOICE_BUILD_CLOUD)
  set(RKVOICE_CLOUD_SDK "${CMAKE_SOURCE_DIR}/build/cloud-sdk" CACHE PATH "ARM64 TLS and Opus SDK")
  set(USE_TLS ON CACHE BOOL "" FORCE)
  set(USE_ZLIB OFF CACHE BOOL "" FORCE)
  set(IXWEBSOCKET_INSTALL OFF CACHE BOOL "" FORCE)
  set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  if(CMAKE_CROSSCOMPILING)
    set(OPENSSL_FOUND TRUE)
    set(OPENSSL_INCLUDE_DIR "${RKVOICE_CLOUD_SDK}/usr/include")
    set(OPENSSL_LIBRARIES "${RKVOICE_CLOUD_SDK}/usr/lib/aarch64-linux-gnu/libssl.so;${RKVOICE_CLOUD_SDK}/usr/lib/aarch64-linux-gnu/libcrypto.so")
    set(OPUS_LIBRARY "${RKVOICE_CLOUD_SDK}/usr/lib/aarch64-linux-gnu/libopus.so")
    set(OPUS_INCLUDE_DIR "${RKVOICE_CLOUD_SDK}/usr/include/opus")
  else()
    find_library(OPUS_LIBRARY NAMES opus libopus.so.0 REQUIRED)
    find_path(OPUS_INCLUDE_DIR NAMES opus.h PATH_SUFFIXES opus)
    if(NOT OPUS_INCLUDE_DIR)
      message(FATAL_ERROR "Install libopus-dev or set OPUS_INCLUDE_DIR to the directory containing opus.h")
    endif()
  endif()
  add_subdirectory(third_party/ixwebsocket EXCLUDE_FROM_ALL)
  set(ENABLE_EXAMPLES OFF CACHE BOOL "" FORCE)
  add_subdirectory(third_party/libfvad EXCLUDE_FROM_ALL)
  if(CMAKE_CROSSCOMPILING)
    target_include_directories(ixwebsocket SYSTEM PUBLIC "${RKVOICE_CLOUD_SDK}/usr/include/aarch64-linux-gnu")
  endif()
  add_executable(rkvoice_cloud src/cloud_main.cpp)
  set(RKVOICE_SPEECH_SDK "" CACHE PATH "Optional sherpa-onnx 1.13.8 shared CPU SDK")
  if(RKVOICE_SPEECH_SDK)
    target_compile_definitions(rkvoice_cloud PRIVATE RKVOICE_SPEECH=1)
    target_link_libraries(rkvoice_cloud PRIVATE "${RKVOICE_SPEECH_SDK}/lib/libsherpa-onnx-c-api.so")
    set_target_properties(rkvoice_cloud PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN/../lib/speech")
    add_executable(rkvoice_speech_test tests/test_speech_engine.cpp)
    target_include_directories(rkvoice_speech_test PRIVATE src third_party "${RKVOICE_PULSE_SDK}/usr/include")
    target_link_libraries(rkvoice_speech_test PRIVATE "${RKVOICE_SPEECH_SDK}/lib/libsherpa-onnx-c-api.so")
    if(CMAKE_CROSSCOMPILING)
      target_link_libraries(rkvoice_speech_test PRIVATE "${PULSE_LIB_DIR}/libpulse-simple.so" "${PULSE_LIB_DIR}/libpulse.so" fvad)
      target_link_options(rkvoice_speech_test PRIVATE "-Wl,--allow-shlib-undefined")
      set_target_properties(rkvoice_speech_test PROPERTIES SKIP_BUILD_RPATH TRUE)
    else()
      target_link_libraries(rkvoice_speech_test PRIVATE PkgConfig::PULSE fvad)
    endif()
  endif()
  target_include_directories(rkvoice_cloud SYSTEM PRIVATE third_party "${OPUS_INCLUDE_DIR}")
  target_compile_options(rkvoice_cloud PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
  target_link_libraries(rkvoice_cloud PRIVATE ixwebsocket ${OPUS_LIBRARY} fvad)
  if(CMAKE_CROSSCOMPILING)
    target_include_directories(rkvoice_cloud SYSTEM PRIVATE "${RKVOICE_PULSE_SDK}/usr/include")
    target_link_libraries(rkvoice_cloud PRIVATE "${PULSE_LIB_DIR}/libpulse-simple.so" "${PULSE_LIB_DIR}/libpulse.so")
    target_link_options(rkvoice_cloud PRIVATE "-Wl,--allow-shlib-undefined")
    set_target_properties(rkvoice_cloud PROPERTIES SKIP_BUILD_RPATH TRUE)
  else()
    target_link_libraries(rkvoice_cloud PRIVATE PkgConfig::PULSE)
  endif()
  add_executable(rkvoice_cloud_state_test tests/test_cloud_state.cpp)
  add_executable(rkvoice_product_test tests/test_products.cpp)
  target_include_directories(rkvoice_product_test PRIVATE third_party)
  target_compile_options(rkvoice_product_test PRIVATE -Wall -Wextra -UNDEBUG)
  target_include_directories(rkvoice_cloud_state_test PRIVATE src third_party "${OPUS_INCLUDE_DIR}")
  target_link_libraries(rkvoice_cloud_state_test PRIVATE ixwebsocket ${OPUS_LIBRARY} fvad)
  add_executable(rkvoice_playback_test tests/test_live_playback.cpp)
  target_include_directories(rkvoice_playback_test PRIVATE src)
  target_link_libraries(rkvoice_playback_test PRIVATE Threads::Threads)
  foreach(test_target rkvoice_cloud_state_test rkvoice_playback_test)
    if(CMAKE_CROSSCOMPILING)
      target_include_directories(${test_target} PRIVATE "${RKVOICE_PULSE_SDK}/usr/include")
      target_link_libraries(${test_target} PRIVATE "${PULSE_LIB_DIR}/libpulse-simple.so" "${PULSE_LIB_DIR}/libpulse.so")
      target_link_options(${test_target} PRIVATE "-Wl,--allow-shlib-undefined")
      set_target_properties(${test_target} PROPERTIES SKIP_BUILD_RPATH TRUE)
    else()
      target_link_libraries(${test_target} PRIVATE PkgConfig::PULSE)
    endif()
  endforeach()
endif()
